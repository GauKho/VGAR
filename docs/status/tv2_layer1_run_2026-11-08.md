## Tái tạo

```bash
export VGAR_TSG_EXECUTABLE=<TSG>/bin/tree-sitter-graph.exe
export VGAR_TSG_TREE_SITTER_DIR=<TSDIR>          # config.json trỏ tới tree-sitter-python v0.23.6
export PATH="/d/tools/winlibs-x86_64-posix-seh-gcc-16.2.0-mingw-w64ucrt-14.0.0-r3/mingw64/bin:$PATH"

# Có TSG
python scripts/build_graph.py D:/kltn/vgar-demo-shop/vgar-demo-shop out/demo-tsg.json --repo-key demo/shop \
  --revision base --tsg-executable "$VGAR_TSG_EXECUTABLE" --tsg-tree-sitter-dir "$VGAR_TSG_TREE_SITTER_DIR"
python scripts/build_graph.py <REPOS>/pydata__xarray-4629 out/xarray-tsg.json --repo-key pydata/xarray \
  --revision base --tsg-executable "$VGAR_TSG_EXECUTABLE" --tsg-tree-sitter-dir "$VGAR_TSG_TREE_SITTER_DIR"

# Không TSG: phải bỏ hai biến VGAR_TSG_*, vì build_graph.py dùng chúng làm mặc định
env -u VGAR_TSG_EXECUTABLE -u VGAR_TSG_TREE_SITTER_DIR \
  python scripts/build_graph.py <repo> out/<name>-plain.json --repo-key <k> --revision base
```
