```python
def test_repair_unexpected_exception_is_audited_and_reraised(tmp_path, monkeypatch):
    from vgar.mcp.servers import repair_server
    audit = tmp_path / "audit.jsonl"
    monkeypatch.setattr(repair_server, "_audit_logger", AuditLogger(audit))
    monkeypatch.setattr(repair_server, "_load", lambda tool, run_dir: (_ for _ in ()).throw(RuntimeError("boom")))
    monkeypatch.setattr(repair_server, "_tool_error", lambda error: (_ for _ in ()).throw(error))
    with pytest.raises(RuntimeError):
        repair_server.verify_patch("r")
    [entry] = _lines(audit)
    assert (entry["payload"]["status"], entry["payload"]["error_code"]) == ("ERROR", "INTERNAL_ERROR")


def test_repair_audit_hashes_inputs_not_raw_text(tmp_path, monkeypatch):
    from vgar.mcp.servers import repair_server
    audit = tmp_path / "audit.jsonl"
    monkeypatch.setattr(repair_server, "_audit_logger", AuditLogger(audit))
    monkeypatch.delenv("VGAR_RUN_ROOT", raising=False)
    repair_server.apply_patch("r", "diff --git a/x b/x\n+SENTINEL_TEXT")
    [entry] = _lines(audit)
    assert "SENTINEL_TEXT" not in audit.read_text(encoding="utf-8")
    assert len(entry["payload"]["input_sha256"]) == 64 and entry["payload"]["server"] == "repair"
```

- [ ] **Step 3: Implement**
- `_audit_logger = AuditLogger(Path(os.getenv("VGAR_MCP_AUDIT_LOG", "logs/mcp_audit.jsonl")))`.
- `_audited(tool)`: bọc hàm bằng `functools.wraps`, nên chữ ký FastMCP thấy được giữ nguyên. Thân hàm bọc:
  - dùng `inspect.signature(fn).bind(*args, **kwargs)` để lấy input, rồi tính `input_sha256` như `_entry`;
  - gọi `fn`;
  - nếu có exception: ghi audit với `status="ERROR"`, `error_code="INTERNAL_ERROR"`, rồi `raise`;
  - nếu không: ghi audit với `status=result["status"]`, và `error_code=result["reason"]` **chỉ khi** `result` có cờ nội bộ đánh dấu nó đến từ `_tool_error`.
- Cách đánh dấu: `_tool_error` trả dict từ `_ToolErrorResult(dict)`, một subclass rỗng của `dict`. Wrapper kiểm bằng `isinstance`. Kết quả JSON không đổi.
- Logger tra qua `globals()` lúc gọi (`_audit_logger.write(...)` trong thân hàm là đủ, vì tên module-global được tra lúc chạy).
