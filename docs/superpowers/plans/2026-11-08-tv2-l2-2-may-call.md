```python
REPOS = "class UserRepo:\n    def check(self): ...\n\nclass OrderRepo:\n    def check(self): ...\n\nclass CacheRepo:\n    def check(self): ...\n"
SERVICE = "class Service:\n    def __init__(self, repo):\n        self.repo = repo\n\n    def run(self):\n        return self.repo.check()\n"

def _write(root, files): ...          # như tests/test_layer1_facts.py
def _build(root): return PythonGraphBuilder(repo_key="t/repo", repository_revision="rev").build(root)

def _may_calls(document):
    nodes = {n["id"]: n for n in document["nodes"]}
    return sorted((nodes[e["target_id"]]["qualified_name"], e["confidence"], tuple(e["provenance"]["evidence"]))
                  for e in document["edges"] if e["type"] == "MAY_CALL")
```
