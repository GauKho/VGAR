RULE_VERSION = "arch-lift-v1"
SOURCE_TOOL = "vgar-arch-lift"

@dataclass(frozen=True)
class EdgeSpec:
    type: str; source_id: str; target_id: str; confidence: float
    resolution: str; rule: str; properties: dict[str, Any]; evidence: tuple[str, ...] = ()

@dataclass(frozen=True)
class DeploymentFile:
    path: str; data: bytes

@dataclass
class LiftResult:
    nodes: list[dict[str, Any]]; edges: list[EdgeSpec]; report: dict[str, Any]

def lift(repo_key: str, nodes: Sequence[dict], edges: Sequence[dict], *,
         is_test: Callable[[str], bool], exports: Mapping[str, frozenset[str]] = {},
